#include <iostream>
using namespace std;

int main() {
    float numbers[5];
    
    cout << "Enter 5 numbers: " << endl;
    for (int i = 0; i < 5; ++i) {
        cout << "Enter number " << i + 1 << ": ";
        cin >> numbers[i];
    }
    float highest = numbers[0];
    
    for (int i = 1; i < 5; ++i) {
        if (highest < numbers[i]) {
            highest = numbers[i];
        }
    }
    
    cout << "\nThe highest number is: " << highest << endl;
    
    return 0;
}
